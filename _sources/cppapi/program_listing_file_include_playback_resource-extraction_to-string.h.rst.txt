   #pragma once
   
   #include "api-types/to-string.h"
   #include "resource-extractor.h"
   
   namespace gpa {
   
   template<>
   inline std::string ToString<gpa::playback::ResourceExtractor::Request::Execution>(gpa::playback::ResourceExtractor::Request::Execution const& value, StrFmtFlags strFmtFlags, uint32_t tabCount, uint32_t tabSize)
   {
       using namespace detail;
       using namespace gpa::playback;
       switch (value) {
       case ResourceExtractor::Request::Execution::PreCall:
           return WriteEnumStr(strFmtFlags, tabCount, tabSize, "PreCall", value);
       case ResourceExtractor::Request::Execution::PostCall:
           return WriteEnumStr(strFmtFlags, tabCount, tabSize, "PostCall", value);
       default:
           return WriteEnumStr(strFmtFlags, tabCount, tabSize, "UNKNOWN", value);
       }
   }
   
   template<>
   inline std::string ToString(gpa::playback::ResourceExtractor::Request::Info const& obj, StrFmtFlags strFmtFlags, uint32_t tabCount, uint32_t tabSize)
   {
       using namespace gpa::detail;
       auto strStr = BeginStructureStr(strFmtFlags, tabCount, tabSize);
       WriteStructureFieldStr(strStr, strFmtFlags, tabCount, tabSize, "callIndex", obj.callIndex) << "," << NewLineStr(strFmtFlags);
       WriteStructureFieldStr(strStr, strFmtFlags, tabCount, tabSize, "stage", obj.stage) << "," << NewLineStr(strFmtFlags);
       if (obj.pResourceInfo) {
           WriteStructureFieldStr(strStr, strFmtFlags, tabCount, tabSize, "pResourceInfo", *obj.pResourceInfo) << "," << NewLineStr(strFmtFlags);
       }
       if (obj.pResourceInfoEx) {
           WriteStructureFieldStr(strStr, strFmtFlags, tabCount, tabSize, "pResourceInfoEx", *obj.pResourceInfoEx) << NewLineStr(strFmtFlags);
       }
       return EndStructureStr(strStr, strFmtFlags, tabCount, tabSize);
   }
   
   template<>
   inline std::string ToString(gpa::playback::ResourceExtractor::Request const& obj, StrFmtFlags strFmtFlags, uint32_t tabCount, uint32_t tabSize)
   {
       using namespace gpa::detail;
       auto strStr = BeginStructureStr(strFmtFlags, tabCount, tabSize);
       WriteStructureFieldStr(strStr, strFmtFlags, tabCount, tabSize, "infoCount", obj.infoCount) << "," << NewLineStr(strFmtFlags);
       WriteArrayFieldStr(strStr, strFmtFlags, tabCount, tabSize, "pInfos", obj.infoCount, obj.pInfos);
       return EndStructureStr(strStr, strFmtFlags, tabCount, tabSize);
   }
